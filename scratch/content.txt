                        <div className="flex flex-col flex-1 min-h-0">
                            <div className="p-4 px-8 border-b border-slate-200 flex items-center justify-between bg-white/20">
                                <div className="flex items-center gap-6">
                                    <div className="flex flex-col">
                                        <span className="font-black text-slate-800 text-lg tracking-tight">พรีวิวข้อมูลนำเข้า</span>
                                        <span className="text-[10px] font-black text-slate-400 uppercase tracking-widest">ตรวจพบทั้งหมด {students.length} รายการ</span>
                                    </div>

                                    {!importSuccess && (
                                        <motion.button
                                            whileHover={{ scale: 1.02 }}
                                            whileTap={{ scale: 0.98 }}
                                            disabled={isCommitting}
                                            onClick={commitImport}
                                            className="flex items-center gap-2 px-6 py-1.5 bg-[#f2f2f7] hover:bg-[#e5e5ea] text-blue-600 rounded-full transition-all font-bold text-[13px] disabled:opacity-30"
                                        >
                                            {isCommitting ? <Loader2 className="animate-spin" size={15} strokeWidth={3} /> : <UploadCloud size={15} strokeWidth={3} />}
                                            {isCommitting ? 'กำลังนำเข้า...' : 'นำเข้าข้อมูล'}
                                        </motion.button>
                                    )}

                                    {importSuccess && (
                                        <div className="flex items-center gap-2 text-emerald-600 bg-emerald-50 px-4 py-2 rounded-full border border-emerald-100 animate-in fade-in zoom-in">
                                            <CheckCircle2 size={16} />
                                            <span className="text-xs font-black uppercase tracking-widest">นำเข้าสำเร็จแล้ว</span>
                                        </div>
                                    )}
                                </div>

                                <button 
                                    onClick={() => {
                                        setStudents([]);
                                        setImportSuccess(false);
                                    }} 
                                    className="w-10 h-10 flex items-center justify-center hover:bg-rose-50 hover:text-rose-500 text-slate-400 rounded-full transition-all border border-transparent hover:border-rose-100"
                                    title="ล้างข้อมูล"
                                >
                                    <X size={20} />
                                </button>
                            </div>

                            <div className="flex-1 overflow-auto p-8 scrollbar-hide">
                                <table className="w-full text-left border-collapse">
                                    <thead>
                                        <tr className="text-[10px] uppercase tracking-[0.2em] text-slate-400 font-black border-b border-slate-100">
                                            <th className="pb-4 pl-4">สถานะ</th>
                                            <th className="pb-4">เลขประจำตัว</th>
                                            <th className="pb-4">ชื่อ-นามสกุล</th>
                                            <th className="pb-4">เบอร์โทรศัพท์</th>
                                            <th className="pb-4 text-right pr-4">อีเมล</th>
                                        </tr>
                                    </thead>
                                    <tbody className="text-sm font-bold text-slate-700">
                                        <AnimatePresence>
                                            {students.map((student) => (
                                                <motion.tr
                                                    key={student.id}
                                                    initial={{ opacity: 0, x: -10 }}
                                                    animate={{ opacity: 1, x: 0 }}
                                                    className="h-16 border-b border-slate-50 hover:bg-white/30 transition-colors group"
                                                >
                                                    <td className="pl-4">
                                                        {student.status === 'ready' && <div className="w-3 h-3 rounded-full bg-blue-400 shadow-[0_0_12px_rgba(96,165,250,0.6)]" />}
                                                        {student.status === 'success' && <CheckCircle2 className="text-emerald-500" size={20} />}
                                                        {student.status === 'error' && <AlertCircle className="text-rose-500" size={20} />}
                                                        {student.status === 'pending' && <Loader2 className="text-slate-300 animate-spin" size={20} />}
                                                    </td>
                                                    <td className="font-mono text-slate-400 text-xs tracking-tighter">{student.studentCode}</td>
                                                    <td className="text-slate-800">{student.prefix}{student.firstName} {student.lastName}</td>
                                                    <td className="text-[11px] text-slate-700 font-bold">{student.phone || '-'}</td>
                                                    <td className="text-[11px] text-slate-400 font-medium text-right pr-4">{student.email || '-'}</td>
                                                </motion.tr>
                                            ))}
                                        </AnimatePresence>
                                    </tbody>
                                </table>
                            </div>

                        </div>
                    )}
